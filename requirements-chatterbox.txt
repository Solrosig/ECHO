# Chatterbox requires numpy<2.0.0, which is IRRECONCILABLE with the main environment

#
#   conda create -n echo-cb python=3.11 -y
#   conda activate echo-cb
#   pip install -r requirements-chatterbox.txt
#
# Render clips here, then analyse in the main `echo` env — the WAV corpus and register.csv
# are the interface between environments, so nothing needs to co-exist:
#
#   (echo-cb)  python synth_stimuli.py --engine chatterbox --param-set all --label chatterbox_test \
#                     --purpose "Native emotion conditioning (X2)"
#   (echo)     python build_register.py --build
#   (echo)     python naturalness.py --register research/register.csv
#   (echo)     python emotion_conveyance.py --register research/register.csv --model ser_model
#
# Optional: place per-quadrant emotional reference clips in refs/Q1.wav .. refs/Q4.wav to
# condition VALENCE via style transfer (ECHO_CHATTERBOX_REFS). Without them the engine still
# runs on the `exaggeration` (arousal) channel alone — which is a useful ablation in itself.
