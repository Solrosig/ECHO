# ECHO — OPTIONAL evaluation layer (not needed to run the system itself).
# Install into the main `echo` environment:  pip install -r requirements-eval.txt
#
# WHY THIS FILE EXISTS (empirically discovered 2026-07-29): a plain `pip install` of a new
# engine silently downgraded numpy/torch/transformers and broke the working analysis stack.
# The version windows below are the ones verified to work together; the comments record the
# constraint that pins each one, so a future reader can re-derive them.
#
# The binding constraint chain:
#   kokoro-onnx   requires numpy >= 2.0.2
#   numba (via librosa) requires numpy <= 2.4   ("Numba needs NumPy 2.4 or less")
#   => the ONLY workable window is numpy >= 2.0.2, < 2.5   (resolves to 2.4.6)

numpy>=2.0.2,<2.5           # kokoro-onnx needs >=2.0.2 ; numba/librosa need <=2.4  -> 2.4.6
praat-parselmouth>=0.4.3    # validated Praat backend: F0 + jitter/shimmer/HNR (verified 0.4.7)
librosa>=0.11               # audio loading for the neural scorers (pulls numba, see above)
torch>=2.6                  # UTMOS + dimensional SER          (verified 2.6.0+cpu)
torchaudio>=2.6             # REQUIRED by the UTMOS model itself, not by our code
transformers>=4.30          # dimensional SER; our loader is version-independent (verified 5.2.0)
